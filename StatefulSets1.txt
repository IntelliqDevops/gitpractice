metadata:
  name: nginx-statefulset
  labels:
    app: nginx

  replicas: 2
  serviceName: nginx
  selector:
    matchLabels:
      app: nginx
  template:
    metadata:
      name: nginx-pod
      labels:
        app: nginx
    spec:
      containers:
        - name: ngnx
          image: nginx

            - containerPort: 80
          volumeMounts:
            - name: new-volume
